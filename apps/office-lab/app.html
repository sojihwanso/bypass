-- Server-side Idempotent Grading & Duplicate XP Prevention RPC (For Single-HTML Direct Call)
create or replace function public.submit_quest_attempt(
  p_quest_id text,
  p_program_id text,
  p_score int,
  p_passed boolean,
  p_base_xp int,
  p_bonus_xp int,
  p_submitted_payload jsonb,
  p_feedback jsonb
) returns jsonb
language plpgsql
security definer
as $$
declare
  v_user_id uuid := auth.uid();
  v_prev_completed boolean := false;
  v_prev_best int := 0;
  v_awarded_xp int := 0;
begin
  if v_user_id is null then
    raise exception 'Not authenticated';
  end if;

  select is_completed, best_score
    into v_prev_completed, v_prev_best
    from public.user_progress
   where user_id = v_user_id and quest_id = p_quest_id;

  if p_passed then
    if not coalesce(v_prev_completed, false) then
      v_awarded_xp := p_base_xp + (case when p_score = 100 then p_bonus_xp else 0 end);
    elsif p_score = 100 and coalesce(v_prev_best, 0) < 100 then
      v_awarded_xp := p_bonus_xp;
    end if;
  end if;

  insert into public.user_progress (user_id, quest_id, current_step, best_score, is_completed, saved_state, updated_at)
  values (v_user_id, p_quest_id, 5, p_score, p_passed, p_submitted_payload, now())
  on conflict (user_id, quest_id) do update
    set current_step = 5,
        best_score = greatest(public.user_progress.best_score, excluded.best_score),
        is_completed = public.user_progress.is_completed or excluded.is_completed,
        saved_state = excluded.saved_state,
        updated_at = now();

  insert into public.quest_attempts (user_id, quest_id, score, passed, submitted_payload, feedback)
  values (v_user_id, p_quest_id, p_score, p_passed, p_submitted_payload, p_feedback);

  if v_awarded_xp > 0 then
    update public.profiles
       set total_xp = total_xp + v_awarded_xp
     where id = v_user_id;
  end if;

  return jsonb_build_object(
    'score', p_score,
    'passed', p_passed,
    'awarded_xp', v_awarded_xp
  );
end;
$$;
